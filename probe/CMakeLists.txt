target_include_directories(MhwSrProbe PRIVATE
  "${MHW_SR_NGX_SDK_DIR}/include")
target_compile_definitions(MhwSrProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(MhwSrProbe PRIVATE /W4 /permissive- /guard:cf /utf-8)
target_link_options(MhwSrProbe PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(MhwSrProbe PRIVATE bcrypt minhook)
set_property(TARGET MhwSrProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
